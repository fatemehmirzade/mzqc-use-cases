python quameter_to_mzqc.py --input_file Mtb-120-outlier-metrics.tsv --location_prefix ftp://massive.ucsd.edu/v01/MSV000079012/

#D:\Software on laptop
library(rmzqc)
Mtb = readMZQC("Mtb-120-outlier-metrics.mzQC")

# Build a data frame to grab the goodies from the mzQC
NumberOfFiles <- length(Mtb$runQualities)
Filenames = rep(NA,NumberOfFiles)
XIC50s = rep(NA,NumberOfFiles)
XICFWHMQ2 = rep(NA,NumberOfFiles)
XICHeightQ2 = rep(NA,NumberOfFiles)
TICQuartRTMiddle = rep(NA,NumberOfFiles)
MS1QuartRTMiddle = rep(NA,NumberOfFiles)
MS2QuartRTMiddle = rep(NA,NumberOfFiles)
MS1TICDeltaQuartRatioQ2 = rep(NA,NumberOfFiles)
MS1TICQuartRatioQ2 = rep(NA,NumberOfFiles)
MS1Count = rep(NA,NumberOfFiles)
MS2Count = rep(NA,NumberOfFiles)
MS1DensityQ2 = rep(NA,NumberOfFiles)
MS2DensityQ2 = rep(NA,NumberOfFiles)
MS2PreZ2 = rep(NA,NumberOfFiles)
MS2PreZ3 = rep(NA,NumberOfFiles)
MS1FastRate = rep(NA,NumberOfFiles)
MS2FastRate = rep(NA,NumberOfFiles)
for (index in 1:NumberOfFiles){
    Filenames[index] <- Mtb$runQualities[[index]]$metadata$inputFiles[[1]]$name
    XIC50s[index] <- Mtb$runQualities[[index]]$qualityMetrics[[1]]$value
    XICFWHMQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[2]]$value[[2]]
    XICHeightQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[3]]$value[[2]]
    TICQuartRTMiddle[index] <- Mtb$runQualities[[index]]$qualityMetrics[[5]]$value[[2]] + Mtb$runQualities[[index]]$qualityMetrics[[5]]$value[[3]]
    MS1QuartRTMiddle[index] <- Mtb$runQualities[[index]]$qualityMetrics[[6]]$value[[2]] + Mtb$runQualities[[index]]$qualityMetrics[[6]]$value[[3]]
    MS2QuartRTMiddle[index] <- Mtb$runQualities[[index]]$qualityMetrics[[7]]$value[[2]] + Mtb$runQualities[[index]]$qualityMetrics[[7]]$value[[3]]
    MS1TICDeltaQuartRatioQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[8]]$value[[2]]
    MS1TICQuartRatioQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[9]]$value[[2]]
    MS1Count[index] <- Mtb$runQualities[[index]]$qualityMetrics[[10]]$value
    MS2Count[index] <- Mtb$runQualities[[index]]$qualityMetrics[[11]]$value
    MS1DensityQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[12]]$value[[2]]
    MS2DensityQ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[13]]$value[[2]]
    MS2PreZ2[index] <- Mtb$runQualities[[index]]$qualityMetrics[[14]]$value[[2]][[2]]
    MS2PreZ3[index] <- Mtb$runQualities[[index]]$qualityMetrics[[14]]$value[[2]][[3]]
    MS1FastRate[index] <- Mtb$runQualities[[index]]$qualityMetrics[[16]]$value
    MS2FastRate[index] <- Mtb$runQualities[[index]]$qualityMetrics[[17]]$value
}
MtbDF = data.frame(XIC50s, XICFWHMQ2, XICHeightQ2, TICQuartRTMiddle, MS1QuartRTMiddle, MS2QuartRTMiddle, MS1TICDeltaQuartRatioQ2, MS1TICQuartRatioQ2,
      MS1Count, MS2Count, MS1DensityQ2, MS2DensityQ2, MS2PreZ2, MS2PreZ3, MS1FastRate, MS2FastRate, row.names=Filenames)

# Run a robust PCA
library(MASS)
library(ggplot2)
library(ggfortify)
robust.cov <- cov.rob(MtbDF)
robust.cor <- cov2cor(robust.cov$cov)
robust.cor.1 <- robust.cov
robust.cor.1$cov <- robust.cor
metrics.pca.1 <- MtbDF
for ( i in 1:ncol(MtbDF)){
  robust.cor.1$center[i] <- robust.cov$center[i]/sqrt(robust.cov$cov[i,i])
  metrics.pca.1[,i] <- MtbDF[,i]/sqrt(robust.cov$cov[i,i])
}

summary(pc.cr <- princomp(metrics.pca.1, covmat = robust.cor.1, scores=T))
loadings<-loadings(pc.cr)  
#plot(pc.cr)
#ggplot2::autoplot(pc.cr,data=pc.cr$scores, scale=0)+geom_text(aes(label=rownames(MtbDF)),hjust=0, vjust=0)

distTable=dist(pc.cr$scores[,1:5], method="euclidean", diag= TRUE,upper=FALSE)
distMatrix=as.matrix(distTable,labels=TRUE)
distDf<-as.data.frame(distMatrix)

Medians = apply(distDf, 1, median, na.rm=T)
names(Medians) = Filenames



# Send this analysis as a setQuality in a new MZQC doc
Rsoftware = toAnalysisSoftware(id = "MS:1000799", version = "2014-02-04", value = "quality control PCA outlier detection", uri = "https://pubs.acs.org/doi/10.1021/ac4034455")
file_format = getCVTemplate(accession = "MS:1003160")
inputFiles = list(MzQCinputFile$new("filename.mzQC", "URI", file_format))
Structure = getQualityMetricTemplate("MS:9999999", getCVSingleton())
Structure$value = list("MS:4000086" = Filename, "MS:4000148" = Medians)
set1_qc = MzQCsetQuality$new(metadata = MzQCmetadata$new(label = "Mtb-120-outlier-pca",
                         inputFiles = inputFiles,
                         analysisSoftware = list(Rsoftware)),
                         qualityMetrics = list(Structure))

mzQC_document = MzQCmzQC$new(version = "1.0.0", 
                             creationDate = MzQCDateTime$new(), 
                             contactName = Sys.info()["user"], 
                             contactAddress = "d.l.tabb@rug.nl", 
                             description = "An example of setQuality used for outliers",
                             setQualities = list(set1_qc),
                             controlledVocabularies = list(getCVInfo()))
writeMZQC("20250116-Mtb-outlier-PCA.mzQC", mzQC_document)
